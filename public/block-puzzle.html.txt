<script>
    const GRID_SIZE = 8;
    
    // Müxtəlif rəng indeksləri ilə fiqur formaları
    const SHAPES = [
        { shape: [[1]], color: 1 },
        { shape: [[1, 1]], color: 2 },
        { shape: [[1], [1]], color: 2 },
        { shape: [[1, 1], [1, 1]], color: 3 },
        { shape: [[1, 1, 1]], color: 4 },
        { shape: [[1], [1], [1]], color: 4 },
        { shape: [[1, 1, 1, 1]], color: 5 },
        { shape: [[1], [1], [1], [1]], color: 5 },
        { shape: [[1, 1], [1, 0]], color: 1 },
        { shape: [[1, 1], [0, 1]], color: 2 },
        { shape: [[1, 1, 1], [1, 0, 0]], color: 3 },
        { shape: [[1, 1, 1], [0, 0, 1]], color: 4 },
        { shape: [[1, 1], [1, 1], [1, 0]], color: 5 }
    ];

    let gameMode = 1;
    let players = [];

    class PlayerGame {
        constructor(id, containerId) {
            this.id = id;
            this.score = 0;
            this.grid = Array(GRID_SIZE).fill(0).map(() => Array(GRID_SIZE).fill(0));
            // Hər hüceyrədə rəng indeksi saxlanılacaq (0 = boş, 1-5 = rəng)
            this.currentPieces = [null, null, null];
            this.selectedPieceIndex = null;
            this.container = document.getElementById(containerId);
            this.isGameOver = false;
            
            this.initDOM();
            this.generateNewPieces();
        }

        initDOM() {
            this.container.innerHTML = `
                <div class="player-panel">
                    <div class="player-info">
                        <div class="player-title">Oyunçu ${this.id}</div>
                        <div class="score-badge">Xal: <span id="score-${this.id}">0</span></div>
                    </div>
                    <div class="grid" id="grid-${this.id}"></div>
                    <div class="pieces-container" id="pieces-${this.id}"></div>
                </div>
            `;
            this.renderGrid();
        }

        renderGrid() {
            const gridEl = document.getElementById(`grid-${this.id}`);
            gridEl.innerHTML = '';
            for (let r = 0; r < GRID_SIZE; r++) {
                for (let c = 0; c < GRID_SIZE; c++) {
                    const cell = document.createElement('div');
                    cell.classList.add('cell');
                    
                    const val = this.grid[r][c];
                    if (val > 0) {
                        cell.classList.add(`filled-${val}`);
                    }

                    // Hover və klik hadisələri
                    cell.addEventListener('mouseenter', () => this.handleCellHover(r, c, true));
                    cell.addEventListener('mouseleave', () => this.handleCellHover(r, c, false));
                    cell.addEventListener('click', () => this.handleCellClick(r, c));

                    gridEl.appendChild(cell);
                }
            }
        }

        generateNewPieces() {
            this.currentPieces = [];
            for (let i = 0; i < 3; i++) {
                const randomObj = SHAPES[Math.floor(Math.random() * SHAPES.length)];
                // Deep copy
                this.currentPieces.push({
                    shape: randomObj.shape.map(row => [...row]),
                    color: randomObj.color
                });
            }
            this.selectedPieceIndex = null;
            this.renderPieces();
            this.checkGameOver();
        }

        renderPieces() {
            const container = document.getElementById(`pieces-${this.id}`);
            container.innerHTML = '';
            
            this.currentPieces.forEach((pieceData, index) => {
                const slot = document.createElement('div');
                slot.classList.add('piece-slot');
                if (this.selectedPieceIndex === index) {
                    slot.classList.add('selected');
                }

                if (pieceData) {
                    const miniGrid = document.createElement('div');
                    miniGrid.classList.add('mini-grid');
                    const shape = pieceData.shape;
                    const rows = shape.length;
                    const cols = shape[0].length;
                    miniGrid.style.gridTemplateColumns = `repeat(${cols}, 16px)`;
                    
                    shape.forEach(row => {
                        row.forEach(val => {
                            const mCell = document.createElement('div');
                            mCell.classList.add('mini-cell');
                            if (val === 1) {
                                mCell.classList.add('filled', `filled-${pieceData.color}`);
                            } else {
                                mCell.classList.add('empty');
                            }
                            miniGrid.appendChild(mCell);
                        });
                    });

                    slot.appendChild(miniGrid);
                    slot.addEventListener('click', (e) => {
                        e.stopPropagation();
                        if (this.isGameOver) return;
                        this.selectedPieceIndex = (this.selectedPieceIndex === index) ? null : index;
                        this.renderPieces();
                    });
                } else {
                    slot.style.opacity = '0.3';
                    slot.style.cursor = 'default';
                }

                container.appendChild(slot);
            });
        }

        handleCellHover(startR, startC, isEnter) {
            if (this.selectedPieceIndex === null || this.isGameOver) return;
            const pieceData = this.currentPieces[this.selectedPieceIndex];
            if (!pieceData) return;

            const shape = pieceData.shape;
            const gridEl = document.getElementById(`grid-${this.id}`);
            const cells = gridEl.children;

            let canPlace = true;
            let targetCells = [];

            for (let r = 0; r < shape.length; r++) {
                for (let c = 0; c < shape[0].length; c++) {
                    if (shape[r][c] === 1) {
                        let curR = startR + r;
                        let curC = startC + c;
                        if (curR >= GRID_SIZE || curC >= GRID_SIZE || this.grid[curR][curC] !== 0) {
                            canPlace = false;
                        } else {
                            targetCells.push(curR * GRID_SIZE + curC);
                        }
                    }
                }
            }

            if (isEnter && canPlace) {
                targetCells.forEach(idx => {
                    if (cells[idx]) cells[idx].classList.add('hover-preview');
                });
            } else {
                for (let i = 0; i < cells.length; i++) {
                    cells[i].classList.remove('hover-preview');
                }
            }
        }

        handleCellClick(startR, startC) {
            if (this.selectedPieceIndex === null || this.isGameOver) return;
            const pieceData = this.currentPieces[this.selectedPieceIndex];
            if (!pieceData) return;

            const shape = pieceData.shape;

            if (this.canPlace(shape, startR, startC)) {
                // Fiquru yerləşdir
                for (let r = 0; r < shape.length; r++) {
                    for (let c = 0; c < shape[0].length; c++) {
                        if (shape[r][c] === 1) {
                            this.grid[startR + r][startC + c] = pieceData.color;
                        }
                    }
                }

                // Xal əlavə et
                let blockCount = shape.flat().reduce((a, b) => a + b, 0);
                this.score += blockCount * 10;

                // Fiquru sıfırla
                this.currentPieces[this.selectedPieceIndex] = null;
                this.selectedPieceIndex = null;

                // Sətir və sütunları təmizlə
                this.clearLines();

                // Yenilə
                document.getElementById(`score-${this.id}`).innerText = this.score;
                this.renderGrid();
                this.renderPieces();

                // Əgər 3 blok da bitdisə, yenilərini gətir
                if (this.currentPieces.every(p => p === null)) {
                    this.generateNewPieces();
                } else {
                    this.checkGameOver();
                }
            }
        }

        canPlace(shape, startR, startC) {
            for (let r = 0; r < shape.length; r++) {
                for (let c = 0; c < shape[0].length; c++) {
                    if (shape[r][c] === 1) {
                        let curR = startR + r;
                        let curC = startC + c;
                        if (curR >= GRID_SIZE || curC >= GRID_SIZE || this.grid[curR][curC] !== 0) {
                            return false;
                        }
                    }
                }
            }
            return true;
        }

        clearLines() {
            let rowsToClear = [];
            let colsToClear = [];

            for (let r = 0; r < GRID_SIZE; r++) {
                if (this.grid[r].every(cell => cell !== 0)) {
                    rowsToClear.push(r);
                }
            }

            for (let c = 0; c < GRID_SIZE; c++) {
                let colFull = true;
                for (let r = 0; r < GRID_SIZE; r++) {
                    if (this.grid[r][c] === 0) colFull = false;
                }
                if (colFull) colsToClear.push(c);
            }

            let clearedCount = rowsToClear.length + colsToClear.length;

            rowsToClear.forEach(r => {
                this.grid[r].fill(0);
            });

            colsToClear.forEach(c => {
                for (let r = 0; r < GRID_SIZE; r++) {
                    this.grid[r][c] = 0;
                }
            });

            if (clearedCount > 0) {
                this.score += clearedCount * 100 * clearedCount; // Bonus xal
            }
        }

        checkGameOver() {
            // Qalan fiqurlardan heç olmasa biri istənilən boş yerə sığışırmı?
            let possible = false;
            this.currentPieces.forEach(pieceData => {
                if (!pieceData) return;
                const shape = pieceData.shape;
                for (let r = 0; r <= GRID_SIZE - shape.length; r++) {
                    for (let c = 0; c <= GRID_SIZE - shape[0].length; c++) {
                        if (this.canPlace(shape, r, c)) {
                            possible = true;
                        }
                    }
                }
            });

            if (!possible && this.currentPieces.some(p => p !== null)) {
                this.isGameOver = true;
                checkAllPlayersGameOver();
            }
        }
    }

    function startGame(mode) {
        gameMode = mode;
        document.getElementById('mainMenu').classList.add('hidden');
        document.getElementById('gameOverModal').classList.add('hidden');
        document.getElementById('gameScreen').classList.remove('hidden');
        
        const wrapper = document.getElementById('boardsWrapper');
        wrapper.innerHTML = '<div id="board1"></div>' + (gameMode === 2 ? '<div id="board2"></div>' : '');

        players = [new PlayerGame(1, 'board1')];
        if (gameMode === 2) {
            players.push(new PlayerGame(2, 'board2'));
        }
    }

    function restartGame() {
        startGame(gameMode);
    }

    function returnToMenu() {
        document.getElementById('gameScreen').classList.add('hidden');
        document.getElementById('gameOverModal').classList.add('hidden');
        document.getElementById('mainMenu').classList.remove('hidden');
    }

    function checkAllPlayersGameOver() {
        if (gameMode === 1) {
            if (players[0].isGameOver) {
                showGameOverModal(`Topladığınız xal: ${players[0].score}`);
            }
        } else {
            // İki nəfərlik rejim
            if (players.every(p => p.isGameOver)) {
                let text = `Oyunçu 1 xalı: ${players[0].score}<br>Oyunçu 2 xalı: ${players[1].score}<br><br>`;
                if (players[0].score > players[1].score) {
                    text += "🏆 Oyunçu 1 Qalib gəldi!";
                } else if (players[1].score > players[0].score) {
                    text += "🏆 Oyunçu 2 Qalib gəldi!";
                } else {
                    text += "🤝 Heç-heçə!";
                }
                showGameOverModal(text);
            }
        }
    }

    function showGameOverModal(message) {
        document.getElementById('gameOverText').innerHTML = message;
        document.getElementById('gameOverModal').classList.remove('hidden');
    }
</script>
